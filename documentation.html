Structures Réelles du Nœud Validateur & Transactions Nexus Core L1 Mainnet #9012 • Base58 • secp256k1 • BigInt microA 
// Constantes réelles du nœud Nexus Core L1
const HTTP_PORT = process.env.HTTP_PORT || 9545;
const CHAIN_ID = 9012;
const COIN_SYMBOL = '₳';
const DECIMALS = 6;
const ONE_A_IN_MICRO = 1_000_000n;
const BLOCK_TIME_MS = 2000;
const MAX_MEMPOOL_SIZE = 50000;

// Structure d'un bloc forgé par le validateur PoA
interface NexusBlock {
  index: number;
  hash: string;          // Base58(SHA256(index + previousHash + timestamp + merkleRoot + validator))
  previousHash: string | null;
  timestamp: number;     // secondes Unix (Math.floor(Date.now() / 1000))
  validator: string;
  transactions: NexusTransaction[];
  merkleRoot: string;    // 'genesis' pour le bloc 0, sinon racine Base58 SHA-256
}

// Structure d'une transaction vérifiée par isValidTransaction(tx)
interface NexusTransaction {
  type?: 'transfer' | 'deploy_contract' | 'call_contract' | 'migration' | 'cloud_mining_reward';
  from: string;          // Adresse Base58 dérivée de publicKey ("ADC" + Base58(SHA256(SHA256(pubKey))), 44 chars)
  to?: string;           // Adresse "ADC..." ou contrat "ADCC..."
  amountMicro: string | number | bigint;
  feeMicro?: string | number | bigint;
  nonce: number;
  publicKey: string;     // Clé publique hex secp256k1
  hash: string;          // Base58(SHA256(JSON.stringify(txData)))
  signature: string;     // Signature DER hex de SHA256(tx.hash) avec la clé privée secp256k1
}
