Base58 + ECDSA secp256k1 
const crypto = require('crypto');
const secp256k1 = require('secp256k1');

const BASE58_ALPHABET = '123456789ABCDEFGHJKLMNPQRSTUVWXYZabcdefghijkmnopqrstuvwxyz';

function encodeBase58(buffer) {
  let digits = [0];
  for (let i = 0; i < buffer.length; i++) {
    let carry = buffer[i];
    for (let j = 0; j < digits.length; ++j) {
      carry += digits[j] << 8;
      digits[j] = carry % 58;
      carry = (carry / 58) | 0;
    }
    while (carry > 0) {
      digits.push(carry % 58);
      carry = (carry / 58) | 0;
    }
  }
  return digits.reverse().map(d => BASE58_ALPHABET[d]).join('');
}

async function broadcastSignedTransfer({ privateKeyHex, fromBase58, toBase58, amountA, nonce, memo = '' }) {
  const chainId = 9012;
  const amountMicro = Math.round(amountA * 1000000);
  const feeMicro = 1000;
  const timestamp = Date.now();

  const canonicalPreImage = `${chainId}:transfer:${fromBase58}:${toBase58}:${amountMicro}:${feeMicro}:${nonce}:${timestamp}`;
  const msgHash = crypto.createHash('sha256').update(canonicalPreImage).digest();
  const sigObj = secp256k1.ecdsaSign(msgHash, Buffer.from(privateKeyHex, 'hex'));
  const signatureBase58 = encodeBase58(Buffer.from(sigObj.signature));

  const response = await fetch('https://rpc.agi-nexus-core.click', {
    method: 'POST',
    headers: { 'Content-Type': 'application/json' },
    body: JSON.stringify({
      jsonrpc: '2.0',
      method: 'nexus_sendRawTransaction',
      params: [{
        chainId,
        type: 'transfer',
        from: fromBase58,
        to: toBase58,
        amountMicro,
        feeMicro,
        nonce,
        timestamp,
        memo,
        signature: signatureBase58
      }],
      id: timestamp
    })
  });
  return await response.json();
}
package bf.atlasduc.nexus.bridge;

import android.webkit.JavascriptInterface;
import org.bitcoinj.core.Base58;
import org.bitcoinj.core.ECKey;
import org.bitcoinj.core.Sha256Hash;
import java.nio.charset.StandardCharsets;

public class NexusCoreWebViewBridge {
    private final ECKey walletKey;
    private final String walletAddressBase58;

    public NexusCoreWebViewBridge(ECKey key, String addressBase58) {
        this.walletKey = key;
        this.walletAddressBase58 = addressBase58;
    }

    @JavascriptInterface
    public String signNexusTransaction(String toBase58, long amountMicro, long feeMicro, long nonce, String memo) {
        long timestamp = System.currentTimeMillis();
        String canonical = "9012:transfer:" + walletAddressBase58 + ":" + toBase58 + ":"
                         + amountMicro + ":" + feeMicro + ":" + nonce + ":" + timestamp;

        Sha256Hash hash = Sha256Hash.of(canonical.getBytes(StandardCharsets.UTF_8));
        ECKey.ECDSASignature sig = walletKey.sign(hash);
        String sigBase58 = Base58.encode(sig.encodeToDER());

        return "{\"chainId\":9012,\"type\":\"transfer\",\"from\":\"" + walletAddressBase58 + "\","
             + "\"to\":\"" + toBase58 + "\",\"amountMicro\":" + amountMicro + ",\"feeMicro\":" + feeMicro + ","
             + "\"nonce\":" + nonce + ",\"timestamp\":" + timestamp + ",\"memo\":\"" + memo + "\","
             + "\"signature\":\"" + sigBase58 + "\"}";
    }
}
